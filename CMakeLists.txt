cmake_minimum_required(VERSION 3.16)
project(gs C)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# SDL2. On a desktop it is built from source and linked in statically. With GS_SYSTEM_SDL2 (the handheld build) gs links against
# libSDL2-2.0.so.0 by name and uses the copy the handheld's system ships: that SDL2 knows the device's GPU, its controller mappings, and
# the frontend's overlays (volume, brightness, battery), and it is updated with the system. The copy built here only serves the linker.
option(GS_SYSTEM_SDL2 "Use the system's libSDL2-2.0.so.0 at run time instead of linking SDL2 in" OFF)
set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
if(GS_SYSTEM_SDL2)
  set(SDL_SHARED ON CACHE BOOL "" FORCE)
  set(SDL_STATIC OFF CACHE BOOL "" FORCE)
else()
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(third_party/sdl2 EXCLUDE_FROM_ALL)

# Lua 5.4 (everything except the standalone interpreter/compiler)
file(GLOB LUA_SRC third_party/lua/src/*.c)
list(FILTER LUA_SRC EXCLUDE REGEX "/(lua|luac)\\.c$")
add_library(lua STATIC ${LUA_SRC})
target_include_directories(lua PUBLIC third_party/lua/src)
if(UNIX AND NOT APPLE)
  target_compile_definitions(lua PRIVATE LUA_USE_LINUX)
elseif(APPLE)
  target_compile_definitions(lua PRIVATE LUA_USE_MACOSX)
endif()

# Embed the Lua framework into the binary as one C byte array: every file in engine/gs/ becomes a module (gs.world, gs.entity, ...), plus the
# built-in instruments and console specs as data modules. The runtime runs it when a cartridge has a game.lua.
function(embed_file in symbol outvar)
  file(READ "${in}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," arr "${hex}")
  set(out "${CMAKE_BINARY_DIR}/embedded_${symbol}.c")
  file(WRITE "${out}" "const unsigned char ${symbol}[] = {${arr}0x00};\nconst unsigned long ${symbol}_len = sizeof(${symbol}) - 1;\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${in}")
  set(${outvar} "${out}" PARENT_SCOPE)
endfunction()
file(GLOB GS_MODULES "${CMAKE_SOURCE_DIR}/engine/gs/*.lua")
list(SORT GS_MODULES)
set(GS_BUNDLE "")
foreach(m ${GS_MODULES})
  get_filename_component(modname "${m}" NAME_WE)
  file(READ "${m}" modsrc)
  string(APPEND GS_BUNDLE "package.preload[\"gs.${modname}\"] = function(...)\n${modsrc}\nend\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${m}")
endforeach()
file(READ "${CMAKE_SOURCE_DIR}/engine/instruments.json" GS_INSTRUMENTS)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/engine/instruments.json")
string(APPEND GS_BUNDLE "package.preload[\"gs.instruments_data\"] = function() return [==[${GS_INSTRUMENTS}]==] end\n")
file(GLOB GS_CONSOLES "${CMAKE_SOURCE_DIR}/engine/consoles/*.json")
list(SORT GS_CONSOLES)
set(GS_CONSOLE_TABLE "")
foreach(c ${GS_CONSOLES})
  get_filename_component(cname "${c}" NAME_WE)
  file(READ "${c}" csrc)
  string(APPEND GS_CONSOLE_TABLE "${cname} = [==[${csrc}]==],\n")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${c}")
endforeach()
string(APPEND GS_BUNDLE "package.preload[\"gs.console_specs\"] = function() return {\n${GS_CONSOLE_TABLE}} end\n")
file(WRITE "${CMAKE_BINARY_DIR}/gs_framework.lua" "${GS_BUNDLE}")
embed_file("${CMAKE_BINARY_DIR}/gs_framework.lua" gs_framework_lua EMBED_FRAMEWORK)

add_executable(gs runtime/main.c runtime/synth.c runtime/sfbank.c third_party/tlsf/tlsf.c ${EMBED_FRAMEWORK})
# tlsf and Lua's own headers: the console's RAM (runtime/ram.c, GS_RAM=1) is built into the program too
target_include_directories(gs PRIVATE third_party/tsf third_party/stb third_party/tlsf third_party/lua/src runtime)
if(GS_SYSTEM_SDL2)
  target_link_libraries(gs PRIVATE lua SDL2::SDL2)
else()
  target_link_libraries(gs PRIVATE lua SDL2::SDL2-static)
endif()
if(NOT WIN32)
  target_link_libraries(gs PRIVATE m ${CMAKE_DL_LIBS})
endif()

# gs as a libretro core (gs_libretro.so / .dylib): the same runtime without its main(), for NextUI's minarch and RetroArch. On the handheld
# it links the system's SDL2, which the emulator has loaded already.
if(NOT WIN32)
  add_library(gs_libretro SHARED runtime/libretro.c runtime/synth.c runtime/sfbank.c third_party/tlsf/tlsf.c ${EMBED_FRAMEWORK})
  set_target_properties(gs_libretro PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden)
  target_include_directories(gs_libretro PRIVATE third_party/tsf third_party/stb third_party/libretro third_party/tlsf third_party/lua/src runtime)
  set_property(TARGET lua PROPERTY POSITION_INDEPENDENT_CODE ON)
  if(GS_SYSTEM_SDL2)
    target_link_libraries(gs_libretro PRIVATE lua SDL2::SDL2 m dl)
  else()
    set_property(TARGET SDL2-static PROPERTY POSITION_INDEPENDENT_CODE ON)
    target_link_libraries(gs_libretro PRIVATE lua SDL2::SDL2-static m)
  endif()
  # the tests' stand-in for minarch (tests/run.sh): loads the core and runs a cart
  if(NOT CMAKE_CROSSCOMPILING)
    add_executable(retro-run tests/retro-run.c)
    target_include_directories(retro-run PRIVATE third_party/libretro)
    target_link_libraries(retro-run PRIVATE ${CMAKE_DL_LIBS})
  endif()
endif()
